services:
  db:
    image: postgres:16-alpine
    container_name: life-care-db
    restart: always

    environment:
      POSTGRES_USER: ${POSTGRES_USER}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
      POSTGRES_DB: ${POSTGRES_DB}

    ports:
      - "5433:5432"

    volumes:
      - postgres_data:/var/lib/postgresql/data

  server:
    build:
      context: ./server
      dockerfile: Dockerfile

    container_name: life-care-server

    restart: always

    ports:
      - "5000:5000"

    env_file:
      - ./server/.env

    depends_on:
      - db

    volumes:
      - ./server:/app
      - /app/node_modules

    command: sh -c "npm run db:push && npm run dev"

  client:
    build:
      context: ./client
      dockerfile: Dockerfile

    container_name: life-care-client

    restart: always

    ports:
      - "3000:3000"

    env_file:
      - ./client/.env.local

    depends_on:
      - server

    volumes:
      - ./client:/app
      - /app/node_modules

    command: npm run dev
